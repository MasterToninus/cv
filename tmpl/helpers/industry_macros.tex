~{ macro url(value) -}~
~{{ value | default('') | replace('\\&', '&') | replace('&', '\\&') | replace('\\%', '%') | replace('%', '\\%') | replace('\\#', '#') | replace('#', '\\#') }}~
~{- endmacro }~

~{ macro voice(entry) -}~
\voice{~{{ entry.what }}~}
    {~{{ entry.when | default('') }}~}
    {~{{ entry.where | default('') }}~}
    {~{{ url(entry.url) }}~}
    {~{{ entry.info | default('') }}~}
~{- endmacro }~

~{ macro voices(entries) -}~
~{ for entry in entries -}~
~{{ voice(entry) }}~
~{ endfor }~
~{- endmacro }~

~{ macro activities(entries) -}~
~{ for entry in entries -}~
\longvoice{~{{ entry.Event_Title }}~}
    {~{{ entry.Type }}~}
    {~{{ url(entry.Url) }}~}
    {~{{ entry.Approx_Date }}~}
    {~{{ entry.Organization }}~}
    {~{{ entry.Location }}~}
    {~{{ entry.Country }}~}
    {~{{ entry.Scheduled }}~}
~{ endfor }~
~{- endmacro }~
